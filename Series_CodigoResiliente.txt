// ===== ESTADO GLOBAL =====
// Datos activos (solo UNA fuente a la vez)
let SERIES_LIST = [];
let SERIES_INDEX = {};
let TOTAL_EPISODES = 0;

// Control de fuentes
let ACTIVE_SOURCE_INDEX = 0;        // Índice de la fuente activa actual
let ACTIVE_SOURCE_FILE = null;      // Nombre del archivo activo
let CONSECUTIVE_FAILURES = 0;       // Contador de fallos consecutivos
let LOADED_SOURCES = {};            // Caché de fuentes cargadas { filename: data }
let SOURCE_STATUS = {};             // Estado de cada fuente { filename: 'ok' | 'down' }

// ===== LOGGING =====
function log(level, message, data = null) {
    const timestamp = new Date().toISOString();
    const prefix = {
        'INFO': 'ℹ️',
        'OK': '✅',
        'WARN': '⚠️',
        'ERROR': '❌',
        'SWITCH': '🔄',
        'FAIL': '💥',
        'RECOVER': '🔧'
    }[level] || '•';
    
    console.log(`[${timestamp}] ${prefix} [${level}] ${message}`, data ? JSON.stringify(data) : '');
}

// ===== CARGA DE UN ARCHIVO DE DATOS =====
function loadDataFile(filename) {
    try {
        const jsonPath = path.join(__dirname, filename);
        
        if (!fs.existsSync(jsonPath)) {
            log('WARN', `Archivo no existe: ${filename}`);
            return null;
        }
        
        const rawData = JSON.parse(fs.readFileSync(jsonPath, 'utf8'));
        
        if (!Array.isArray(rawData)) {
            log('WARN', `Formato inválido: ${filename}`);
            return null;
        }

        // Procesar datos
        const map = {};
        const allUrls = [];
        
        rawData.forEach(item => {
            const name = item.series || 'Sin nombre';
            const season = String(item.season || '1');
            
            if (!map[name]) {
                map[name] = { 
                    name, 
                    poster: item["logo serie"] || '', 
                    seasons: {}, 
                    count: 0 
                };
            }
            
            if (!map[name].seasons[season]) {
                map[name].seasons[season] = [];
            }
            
            const url = item.url || '';
            map[name].seasons[season].push({ 
                ep: item.ep || 1, 
                title: item.title || 'Episodio ' + (item.ep || 1), 
                url: url
            });
            map[name].count++;
            
            if (url) allUrls.push(url);
        });

        // Ordenar episodios
        Object.values(map).forEach(s => 
            Object.keys(s.seasons).forEach(k => 
                s.seasons[k].sort((a, b) => a.ep - b.ep)
            )
        );
        
        // Crear lista de series
        const seriesList = Object.values(map)
            .map(s => ({ 
                name: s.name, 
                poster: s.poster, 
                seasons: Object.keys(s.seasons).length, 
                count: s.count 
            }))
            .sort((a, b) => a.name.localeCompare(b.name));

        log('OK', `Cargado: ${filename}`, { series: seriesList.length, episodios: rawData.length });
        
        return {
            seriesIndex: map,
            seriesList: seriesList,
            totalEpisodes: rawData.length,
            sampleUrls: allUrls.slice(0, 10) // Guardar algunas URLs para testing
        };
        
    } catch (e) { 
        log('ERROR', `Error cargando ${filename}: ${e.message}`);
        return null;
    }
}

// ===== ACTIVAR UNA FUENTE =====
function activateSource(index) {
    if (index < 0 || index >= config.DATA_FILES.length) {
        log('ERROR', `Índice de fuente inválido: ${index}`);
        return false;
    }
    
    const filename = config.DATA_FILES[index];
    
    // Si ya está cargada en caché, usarla
    if (!LOADED_SOURCES[filename]) {
        const data = loadDataFile(filename);
        if (!data) {
            SOURCE_STATUS[filename] = 'down';
            return false;
        }
        LOADED_SOURCES[filename] = data;
    }
    
    const sourceData = LOADED_SOURCES[filename];
    
    // Activar esta fuente
    SERIES_INDEX = sourceData.seriesIndex;
    SERIES_LIST = sourceData.seriesList;
    TOTAL_EPISODES = sourceData.totalEpisodes;
    ACTIVE_SOURCE_INDEX = index;
    ACTIVE_SOURCE_FILE = filename;
    CONSECUTIVE_FAILURES = 0;
    SOURCE_STATUS[filename] = 'ok';
    
    log('OK', `Fuente activa: ${filename}`, { 
        series: SERIES_LIST.length, 
        episodios: TOTAL_EPISODES 
    });
    
    return true;
}

// ===== CAMBIAR A SIGUIENTE FUENTE =====
function switchToNextSource() {
    const currentFile = ACTIVE_SOURCE_FILE;
    SOURCE_STATUS[currentFile] = 'down';
    
    log('SWITCH', `Fuente ${currentFile} marcada como caída, buscando alternativa...`);
    
    // Buscar siguiente fuente disponible
    for (let i = ACTIVE_SOURCE_INDEX + 1; i < config.DATA_FILES.length; i++) {
        const filename = config.DATA_FILES[i];
        
        // Saltar si ya está marcada como caída
        if (SOURCE_STATUS[filename] === 'down') {
            log('INFO', `Saltando ${filename} (ya marcada como caída)`);
            continue;
        }
        
        log('INFO', `Intentando activar: ${filename}`);
        
        if (activateSource(i)) {
            log('SWITCH', `✅ Cambiado a: ${filename}`);
            return true;
        }
    }
    
    log('ERROR', 'No hay más fuentes disponibles');
    return false;
}

// ===== REGISTRAR FALLO DE REPRODUCCIÓN =====
function registerPlaybackFailure(url) {
    CONSECUTIVE_FAILURES++;
    
    log('FAIL', `Fallo #${CONSECUTIVE_FAILURES} en ${ACTIVE_SOURCE_FILE}`, { 
        url: url?.substring(0, 50) + '...',
        threshold: config.FAILURE_THRESHOLD 
    });
    
    // Si superamos el umbral, cambiar de fuente
    if (CONSECUTIVE_FAILURES >= config.FAILURE_THRESHOLD) {
        log('WARN', `Umbral de fallos alcanzado (${CONSECUTIVE_FAILURES}), cambiando fuente...`);
        switchToNextSource();
    }
}

// ===== REGISTRAR ÉXITO DE REPRODUCCIÓN =====
function registerPlaybackSuccess() {
    // Resetear contador de fallos en caso de éxito
    if (CONSECUTIVE_FAILURES > 0) {
        log('OK', `Reproducción exitosa, reseteando contador de fallos`);
        CONSECUTIVE_FAILURES = 0;
    }
}

// ===== VERIFICAR URL =====
function testUrl(url, timeout = config.URL_TEST_TIMEOUT) {
    return new Promise((resolve) => {
        try {
            const parsed = new URL(url);
            const client = parsed.protocol === 'https:' ? https : http;
            
            const req = client.request({
                hostname: parsed.hostname,
                port: parsed.port || (parsed.protocol === 'https:' ? 443 : 80),
                path: parsed.pathname + parsed.search,
                method: 'HEAD',
                timeout: timeout,
                headers: { 'User-Agent': 'Mozilla/5.0' }
            }, (res) => {
                resolve(res.statusCode >= 200 && res.statusCode < 400);
            });

            req.on('error', () => resolve(false));
            req.on('timeout', () => { req.destroy(); resolve(false); });
            req.end();
        } catch (e) {
            resolve(false);
        }
    });
}

// ===== INTENTAR RECUPERAR FUENTE PRINCIPAL =====
async function tryRecoverPrimarySource() {
    // Solo intentar si no estamos ya en la fuente principal
    if (ACTIVE_SOURCE_INDEX === 0) return;
    
    const primaryFile = config.DATA_FILES[0];
    
    // Solo intentar si la primaria está marcada como caída
    if (SOURCE_STATUS[primaryFile] !== 'down') return;
    
    log('RECOVER', `Verificando si fuente principal (${primaryFile}) se recuperó...`);
    
    // Obtener URLs de prueba de la fuente principal
    if (!LOADED_SOURCES[primaryFile]) {
        const data = loadDataFile(primaryFile);
        if (!data) return;
        LOADED_SOURCES[primaryFile] = data;
    }
    
    const sampleUrls = LOADED_SOURCES[primaryFile].sampleUrls || [];
    if (sampleUrls.length === 0) return;
    
    // Probar algunas URLs
    const urlsToTest = sampleUrls.slice(0, 3);
    let successCount = 0;
    
    for (const url of urlsToTest) {
        if (await testUrl(url)) {
            successCount++;
        }
    }
    
    // Si al menos 2 de 3 funcionan, recuperar
    if (successCount >= 2) {
        log('RECOVER', `✅ Fuente principal recuperada! (${successCount}/3 URLs OK)`);
        SOURCE_STATUS[primaryFile] = 'ok';
        activateSource(0);
    } else {
        log('INFO', `Fuente principal sigue caída (${successCount}/3 URLs OK)`);
    }
}

// ===== INICIALIZACIÓN =====
function initialize() {
    log('INFO', '========================================');
    log('INFO', 'Iniciando Stream+...');
    log('INFO', `Fuentes configuradas: ${config.DATA_FILES.length}`);
    
    // Intentar activar la fuente principal
    for (let i = 0; i < config.DATA_FILES.length; i++) {
        if (activateSource(i)) {
            break;
        }
        log('WARN', `No se pudo cargar ${config.DATA_FILES[i]}, intentando siguiente...`);
    }
    
    if (!ACTIVE_SOURCE_FILE) {
        log('ERROR', 'No se pudo cargar ninguna fuente de datos');
    }
    
    log('INFO', '========================================');
}

initialize();

// Intentar recuperar fuente principal periódicamente
setInterval(() => {
    tryRecoverPrimarySource();
}, config.RECOVERY_INTERVAL);

app.get('/api/series', (req, res) => {
    const page = parseInt(req.query.page) || 0;
    const limit = parseInt(req.query.limit) || 250;
    const search = (req.query.q || '').toLowerCase();
    const random = req.query.random === 'true';
    
    let list = [...SERIES_LIST];
    if (search) list = list.filter(s => s.name.toLowerCase().includes(search));
    if (random) {
        for (let i = list.length - 1; i > 0; i--) {
            const j = Math.floor(Math.random() * (i + 1));
            [list[i], list[j]] = [list[j], list[i]];
        }
    }
    
    const start = page * limit;
    res.json({ 
        total: list.length, 
        page, 
        hasMore: start + limit < list.length, 
        data: list.slice(start, start + limit) 
    });
});

app.get('/api/series/:name', (req, res) => {
    const series = SERIES_INDEX[decodeURIComponent(req.params.name)];
    if (!series) return res.status(404).json({ error: 'No encontrada' });
    res.json({ data: series });
});

// ===== VIDEO PROXY =====
app.get('/video-proxy', videoProxyLimiter, (req, res) => {
    const url = req.query.url;
    if (!url) return res.status(400).json({ error: 'URL requerida' });

    let parsed;
    try {
        parsed = new URL(decodeURIComponent(url));
    } catch (e) {
        return res.status(400).json({ error: 'URL inválida' });
    }

    const client = parsed.protocol === 'https:' ? https : http;

    const opts = { 
        hostname: parsed.hostname, 
        port: parsed.port || (parsed.protocol === 'https:' ? 443 : 80), 
        path: parsed.pathname + parsed.search, 
        method: 'GET',
        timeout: 30000,
        headers: { 
            'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36',
            'Accept': '*/*',
            'Accept-Encoding': 'identity',
            'Connection': 'keep-alive',
            'Referer': parsed.origin + '/'
        } 
    };

    if (req.headers.range) {
        opts.headers['Range'] = req.headers.range;
    }

    const proxyReq = client.request(opts, proxyRes => {
        // Redirect
        if ([301, 302, 303, 307, 308].includes(proxyRes.statusCode) && proxyRes.headers.location) {
            proxyRes.destroy();
            return res.redirect('/video-proxy?url=' + encodeURIComponent(proxyRes.headers.location));
        }

        // Error del servidor origen
        if (proxyRes.statusCode >= 400) {
            proxyRes.destroy();
            // NO registrar fallo aquí, el cliente lo reportará
            return res.status(proxyRes.statusCode).json({ 
                error: 'Error del servidor origen',
                statusCode: proxyRes.statusCode
            });
        }

        // Éxito
        const headers = {
            'Content-Type': proxyRes.headers['content-type'] || 'video/mp4',
            'Accept-Ranges': 'bytes',
            'Cache-Control': 'public, max-age=3600'
        };

        if (proxyRes.headers['content-length']) headers['Content-Length'] = proxyRes.headers['content-length'];
        if (proxyRes.headers['content-range']) headers['Content-Range'] = proxyRes.headers['content-range'];

        res.writeHead(proxyRes.statusCode, headers);
        proxyRes.pipe(res, { end: true });

        proxyRes.on('error', (err) => {
            if (!res.headersSent) res.status(502).json({ error: 'Stream error' });
            else res.end();
        });
    });

    proxyReq.on('timeout', () => {
        proxyReq.destroy();
        if (!res.headersSent) res.status(504).json({ error: 'Timeout' });
    });

    proxyReq.on('error', (err) => {
        if (!res.headersSent) res.status(502).json({ error: 'Connection error' });
    });

    req.on('close', () => proxyReq.destroy());
    proxyReq.end();
});

                <div class="p-source" id="pSource">--</div>
            </div>
            <div class="p-bottom">
                <div class="p-bar" id="pBar"><div class="p-bar-fill" id="pFill"></div></div>
                <div class="p-ctrl">
                    <button class="p-btn" id="pRw">◀◀</button>
                    <button class="p-btn" id="pPp">▶</button>
                    <button class="p-btn" id="pFw">▶▶</button>
                </div>
            </div>
        </div>
    </div>
</div>
<script>
(function(){
const $=id=>document.getElementById(id);
const state={series:null,seriesName:null,season:null,epIdx:0,ep:null,page:0,hasMore:true,loading:false,retries:0};

const el={
    grid:$('grid'),srch:$('srch'),mix:$('mix'),stats:$('stats'),srcBadge:$('srcBadge'),
    detail:$('detail'),detBack:$('detBack'),detTitle:$('detTitle'),tabs:$('tabs'),eps:$('eps'),
    player:$('player'),vid:$('vid'),pUi:$('pUi'),pTitle:$('pTitle'),pSource:$('pSource'),
    pLoad:$('pLoad'),pLoadTxt:$('pLoadTxt'),pErr:$('pErr'),pErrSub:$('pErrSub'),
    pRetry:$('pRetry'),pClose:$('pClose'),pBar:$('pBar'),pFill:$('pFill'),pRw:$('pRw'),pPp:$('pPp'),pFw:$('pFw')
};

let hideT,activeSource='';

// === Init ===
loadStats();
loadSeries(false,true);
setInterval(loadStats,30000);

function loadStats(){
    fetch('/api/stats').then(r=>r.json()).then(d=>{
        el.stats.textContent=d.series+' series';
        activeSource=d.activeSource;
        const srcNum=d.sourceIndex+'/'+d.totalSources;
        el.srcBadge.textContent=srcNum;
        el.srcBadge.className='source-badge'+(d.consecutiveFailures>0?' warn':' ok');
    }).catch(()=>{});
}

function loadSeries(append,random){
    if(state.loading)return;
    state.loading=true;
    if(!append){el.grid.innerHTML='<div class="msg">Cargando...</div>';state.page=0;}
    
    let u='/api/series?page='+state.page+'&limit=50';
    if(el.srch.value.trim())u+='&q='+encodeURIComponent(el.srch.value);
    if(random)u+='&random=true';
    
    fetch(u).then(r=>r.json()).then(d=>{
        if(!append)el.grid.innerHTML='';
        if(!d.data.length&&!append){el.grid.innerHTML='<div class="msg">Sin resultados</div>';return;}
        d.data.forEach(s=>{
            const c=document.createElement('div');c.className='card';
            c.innerHTML='<img data-src="'+esc(s.poster)+'"><div class="card-t">'+esc(s.name)+'</div>';
            const img=c.querySelector('img');
            obs.observe(img);
            c.onclick=()=>openDetail(s.name);
            el.grid.appendChild(c);
        });
        state.page++;state.hasMore=d.hasMore;
    }).catch(()=>{if(!append)el.grid.innerHTML='<div class="msg">Error</div>';})
    .finally(()=>state.loading=false);
}

const obs=new IntersectionObserver(es=>{
    es.forEach(e=>{
        if(e.isIntersecting){
            const img=e.target;
            if(img.dataset.src){img.src=img.dataset.src;img.onload=()=>img.classList.add('ok');}
            obs.unobserve(img);
        }
    });
},{rootMargin:'100px'});

function openDetail(name){
    state.seriesName=name;
    el.detTitle.textContent=name;
    el.detail.classList.add('open');
    el.tabs.innerHTML='<div class="msg">Cargando...</div>';
    el.eps.innerHTML='';
    
    fetch('/api/series/'+encodeURIComponent(name)).then(r=>r.json()).then(d=>{
        state.series=d.data;
        const seasons=Object.keys(state.series.seasons).sort((a,b)=>a-b);
        state.season=seasons[0];
        
        el.tabs.innerHTML='';
        seasons.forEach(s=>{
            const t=document.createElement('button');
            t.className='tab'+(s===state.season?' on':'');
            t.textContent='T'+s;
            t.onclick=()=>{state.season=s;el.tabs.querySelectorAll('.tab').forEach(x=>x.classList.toggle('on',x.textContent==='T'+s));renderEps();};
            el.tabs.appendChild(t);
        });
        renderEps();
    }).catch(()=>el.tabs.innerHTML='<div class="msg">Error</div>');
}

function renderEps(){
    const eps=state.series?.seasons[state.season]||[];
    el.eps.innerHTML='';
    eps.forEach((ep,i)=>{
        const d=document.createElement('div');d.className='ep';
        d.innerHTML='<div class="ep-n">'+ep.ep+'</div><div class="ep-info"><div class="ep-t">'+esc(ep.title)+'</div><div class="ep-m">Temporada '+state.season+'</div></div>';
        d.onclick=()=>{state.epIdx=i;state.ep=ep;openPlayer(ep);};
        el.eps.appendChild(d);
    });
}

function openPlayer(ep){
    state.ep=ep;state.retries=0;
    el.player.classList.add('open');
    playVideo(ep.url);
}

function playVideo(url){
    el.pErr.classList.remove('show');
    el.pLoad.classList.add('show');
    el.pLoadTxt.textContent='Conectando...';
    el.pTitle.textContent=state.ep.title;
    el.pSource.textContent='Fuente: '+activeSource;
    
    let src=url;
    if(!url.startsWith('https://')){
        src='/video-proxy?url='+encodeURIComponent(url);
    }
    
    el.vid.src=src;
    el.vid.play().catch(()=>{});
}

// === Video Events ===
el.vid.onloadstart=()=>{el.pLoad.classList.add('show');el.pLoadTxt.textContent='Cargando...';};
el.vid.oncanplay=()=>{
    el.pLoad.classList.remove('show');
    state.retries=0;
    // Reportar éxito
    fetch('/api/report-success',{method:'POST',headers:{'Content-Type':'application/json'},body:'{}'}).catch(()=>{});
};
el.vid.onwaiting=()=>{el.pLoad.classList.add('show');el.pLoadTxt.textContent='Buffering...';};
el.vid.onplaying=()=>{el.pLoad.classList.remove('show');el.pPp.textContent='⏸';};
el.vid.onpause=()=>el.pPp.textContent='▶';
el.vid.ontimeupdate=()=>{if(el.vid.duration)el.pFill.style.width=(el.vid.currentTime/el.vid.duration*100)+'%';};
el.vid.onerror=handleVideoError;

function handleVideoError(){
    const err=el.vid.error;
    let msg='Error desconocido';
    if(err){
        switch(err.code){
            case 1:msg='Carga abortada';break;
            case 2:msg='Error de red';break;
            case 3:msg='Error de decodificación';break;
            case 4:msg='Formato no soportado';break;
        }
    }
    
    // Reportar fallo al servidor
    fetch('/api/report-failure',{
        method:'POST',
        headers:{'Content-Type':'application/json'},
        body:JSON.stringify({url:state.ep?.url})
    }).then(r=>r.json()).then(d=>{
        // Actualizar badge si cambió la fuente
        if(d.activeSource!==activeSource){
            activeSource=d.activeSource;
            el.pSource.textContent='Fuente: '+activeSource+' (cambio automático)';
            // Recargar stats
            loadStats();
        }
    }).catch(()=>{});
    
    // Auto-reintentar hasta 3 veces
    if(state.retries<3){
        state.retries++;
        el.pLoadTxt.textContent='Reintentando ('+state.retries+'/3)...';
        setTimeout(()=>playVideo(state.ep.url),2000);
    }else{
        el.pLoad.classList.remove('show');
        el.pErr.classList.add('show');
        el.pErrSub.textContent=msg;
    }
}

// === Player Controls ===
el.pPp.onclick=()=>{if(el.vid.paused)el.vid.play();else el.vid.pause();showUI();};
el.pRw.onclick=()=>{el.vid.currentTime=Math.max(0,el.vid.currentTime-10);showUI();};
el.pFw.onclick=()=>{el.vid.currentTime+=10;showUI();};
el.pBar.onclick=e=>{const r=el.pBar.getBoundingClientRect();el.vid.currentTime=(e.clientX-r.left)/r.width*el.vid.duration;};
el.pRetry.onclick=()=>{state.retries=0;playVideo(state.ep.url);};
el.pClose.onclick=closePlayer;
el.player.onclick=e=>{if(e.target===el.vid){if(el.vid.paused)el.vid.play();else el.vid.pause();showUI();}};
el.player.onmousemove=showUI;

function showUI(){el.pUi.classList.remove('hide');clearTimeout(hideT);hideT=setTimeout(()=>{if(!el.vid.paused)el.pUi.classList.add('hide');},3000);}
function closePlayer(){el.vid.pause();el.vid.src='';el.player.classList.remove('open');}

// === Navigation ===
el.detBack.onclick=()=>el.detail.classList.remove('open');
el.mix.onclick=()=>loadSeries(false,true);
let st;el.srch.oninput=()=>{clearTimeout(st);st=setTimeout(()=>loadSeries(false,!el.srch.value.trim()),300);};
document.getElementById('main').onscroll=function(){
    if(!state.loading&&state.hasMore&&this.scrollTop+this.clientHeight>=this.scrollHeight-200)loadSeries(true,false);
};

function esc(s){return s?String(s).replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'})[c]):''}
})();
</script>
</body>
</html>`;
